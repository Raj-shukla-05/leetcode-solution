class Solution {
public:
    int longestSubsequence(vector<int>& nums) {
        int xorAll = 0;
        int nonZero = 0;
        for (int num : nums) {
            xorAll ^= num;
            if (num != 0) nonZero++;
        }
        if (nonZero == 0) return 0;
        if (xorAll != 0) return nums.size();
        return nums.size() - 1;
    }
};
