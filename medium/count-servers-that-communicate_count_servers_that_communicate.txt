
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                if (grid[i][j]) {
                    row[i]++;
                    col[j]++;
                }

        int ans = 0;
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                if (grid[i][j] && (row[i] > 1 || col[j] > 1))
                    ans++;

        return ans;
    }
