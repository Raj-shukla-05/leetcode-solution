            
            for (int i = 0; i < half.size(); i++) {
                if (used[i]) continue;
                if (i > 0 && half[i] == half[i - 
                1] && !used[i - 1]) continue;
                
                used[i] = true;
                result.push_back(half[i]);
                
                string prefix = result;
                string rev = prefix;
                reverse(rev.begin(), rev.end());
                string fullPrefix = prefix + mid + 
                rev;
                
                if (fullPrefix.substr(0, pos + 1) >= 
                target.substr(0, pos + 1)) {
                    if (dfs(pos + 1)) return true;
                }
                
                result.pop_back();
                used[i] = false;
            }
            
            return false;
        };
        
        if (dfs(0)) {
            string rev = result;
            reverse(rev.begin(), rev.end());
            return result + mid + rev;
        }
        return "";
    }
