            }
        }
        
        vector<int> dp(n + 1, 0);
        
        for (int i = 1; i <= n; i++) {
            dp[i] = dp[i - 1];
            
            for (int j = i - k; j >= 0; j--) {
                if (isPal[j][i - 1]) {
                    dp[i] = max(dp[i], dp[j] + 1);
                    break;
                }
            }
        }
        
        return dp[n];
    }
};
