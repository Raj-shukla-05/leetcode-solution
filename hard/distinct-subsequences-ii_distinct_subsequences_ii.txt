        vector<int> dp(n + 1, 0);
        vector<int> last(26, -1);
        
        dp[0] = 1;
        
        for (int i = 1; i <= n; i++) {
            dp[i] = (2 * dp[i - 1]) % MOD;
            
            int idx = s[i - 1] - 'a';
            if (last[idx] != -1) {
                dp[i] = (dp[i] - dp[last[idx] - 1] + MOD) % MOD;
            }
            
            last[idx] = i;
        }
        
        return (dp[n] - 1 + MOD) % MOD;
    }
};
