class Solution {
public:
    string decodeCiphertext(string encodedText, int rows) {
        int len = encodedText.size();
        if (len == 0 || rows == 1) return encodedText;
        
        int cols = len / rows;
        vector<vector<char>> matrix(rows, vector<char>(cols));
        
        for (int i = 0; i < len; i++) {
            matrix[i / cols][i % cols] = encodedText[i];
        }
        
        string result = "";
        for (int c = 0; c < cols; c++) {
            int r = 0, cc = c;
            while (r < rows && cc < cols) {
                result += matrix[r][cc];
                r++;
                cc++;
            }
        }
        
        while (!result.empty() && result.back() == ' ') {
            result.pop_back();
        }
        
        return result;
    }
};
