                    for (int j = 0; j < n && !canMake; j++) {
                        if (i != j) {
                            int diff = nums1[i] - nums1[j];
                            if ((diff % 2 + 2) % 2 == targetParity) {
                                canMake = true;
                            }
                        }
                    }
                }
                
                if (!canMake) possible = false;
            }
            
            if (possible) return true;
        }
        
        return false;
    }
};
